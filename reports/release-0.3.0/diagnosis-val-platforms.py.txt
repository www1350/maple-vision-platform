def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, ensure_ascii=True, sort_keys=True, indent=2, allow_nan=False)
        stream.write('\n')


def raw_rows(output, transform):
    rows = np.asarray(output[0].T, dtype=np.float64)
    assert rows.shape == (8400, 11)
    valid = np.isfinite(rows).all(axis=1)
    valid &= ((rows[:, 4:] >= 0) & (rows[:, 4:] <= 1)).all(axis=1)
    valid &= (rows[:, 2] > 0) & (rows[:, 3] > 0)
    boxes = np.empty((len(rows), 4), dtype=np.float64)
    boxes[:, 0] = (rows[:, 0] - rows[:, 2] / 2 - transform.pad_x) / transform.scale_x
    boxes[:, 2] = (rows[:, 0] + rows[:, 2] / 2 - transform.pad_x) / transform.scale_x
    boxes[:, 1] = (rows[:, 1] - rows[:, 3] / 2 - transform.pad_y) / transform.scale_y
    boxes[:, 3] = (rows[:, 1] + rows[:, 3] / 2 - transform.pad_y) / transform.scale_y
    boxes[:, (0, 2)] = np.clip(boxes[:, (0, 2)], 0, transform.original_width)
    boxes[:, (1, 3)] = np.clip(boxes[:, (1, 3)], 0, transform.original_height)
    valid &= (boxes[:, 2] > boxes[:, 0]) & (boxes[:, 3] > boxes[:, 1])
    classes = np.argmax(rows[:, 4:], axis=1)
    return rows, boxes, valid, classes


def raw_diagnosis(target, arrays, class_names):
    rows, boxes, valid, classes = arrays
    gt = np.asarray(target.bbox, dtype=np.float64)
    intersections = np.maximum(0, np.minimum(boxes[:, 2:], gt[2:]) - np.maximum(boxes[:, :2], gt[:2]))
    intersection = intersections[:, 0] * intersections[:, 1]
    gt_area = (gt[2] - gt[0]) * (gt[3] - gt[1])
    box_area = (boxes[:, 2] - boxes[:, 0]) * (boxes[:, 3] - boxes[:, 1])
    overlaps = np.divide(intersection, box_area + gt_area - intersection)
    overlaps[~valid] = -1
    platform_index = class_names.index('platform')
    platform = valid & (classes == platform_index)
    scores = rows[:, 4 + platform_index]

    def item(index):
        index = int(index)
        if index < 6400:
            stride, local, grid = 4, index, 80
        elif index < 8000:
            stride, local, grid = 8, index - 6400, 40
        else:
            stride, local, grid = 16, index - 8000, 20
        return {'raw_index':index, 'bbox':boxes[index].tolist(), 'iou':float(overlaps[index]),
                'platform_score':float(scores[index]), 'winning_class':class_names[int(classes[index])],
                'winning_score':float(rows[index, 4 + classes[index]]), 'stride':stride,
                'anchor_model_xy':[(local % grid + .5) * stride, (local // grid + .5) * stride]}

    def best(mask, score=False):
        indices = np.flatnonzero(mask)
        if not len(indices):
            return None
        chosen = indices[np.argmax(scores[indices] if score else overlaps[indices])]
        return item(chosen)

    overlap_valid = valid & (overlaps >= .5)
    platform_overlap = platform & (overlaps >= .5)
    confident_platform = platform & (scores >= .25)
    confident_overlap = confident_platform & (overlaps >= .5)
    if confident_overlap.any():
        absence_reason = 'qualifying_raw_candidate_not_retained_or_not_matched'
    elif platform_overlap.any():
        absence_reason = 'raw_platform_geometry_exists_but_score_below_frozen_threshold'
    elif overlap_valid.any():
        absence_reason = 'iou50_geometry_exists_only_with_another_winning_class'
    else:
        absence_reason = 'no_raw_decoded_box_reaches_iou50'
    return {
        'raw_valid_rows':int(valid.sum()),
        'raw_any_class_iou50_rows':int(overlap_valid.sum()),
        'raw_platform_iou50_rows':int(platform_overlap.sum()),
        'raw_platform_iou50_rows_above_frozen_confidence':int(confident_overlap.sum()),
        'best_any_class_iou':best(valid),
        'best_platform_iou':best(platform),
        'highest_platform_score_at_iou50':best(platform_overlap, score=True),
        'best_platform_iou_above_frozen_confidence':best(confident_platform),
        'highest_platform_score_any_iou':best(platform, score=True),
        'absence_reason_if_unmatched':absence_reason,
    }

ledger=[]

first_miss=None
native_calls=0
for image_index, record in enumerate(records):
    image_path, label_path = DATA / record['image'], DATA / record['label']
    assert digest(image_path) == record['image_sha256']
    assert digest(label_path) == record['label_sha256']
    frame = ImageFrame.from_file(image_path,sequence=record['sequence'],timestamp_ms=record['timestamp_ms'])
    tensor, transform = preprocess(frame, model.manifest.input_width, model.manifest.input_height)
    raw = np.asarray(model._session.run(None, {model._input_name:tensor})[0])
    native_calls += 1
    predictions = decode(raw, model.manifest, transform, confidence_threshold=.25, iou_threshold=.45)
    truth = _ground_truth(label_path, frame.width, frame.height)
    samples.append(EvaluationSample(truth, predictions))
    matched={}
    for pi, prediction in sorted(enumerate(predictions), key=lambda pair:(-pair[1].confidence,pair[0])):
        if prediction.confidence < .25:
            continue
        options=[(_iou(prediction,target),ti) for ti,target in enumerate(truth)
                 if target.class_id==prediction.class_id and ti not in matched]
        best=max(options,key=lambda pair:(pair[0],-pair[1]),default=None)
        if best is not None and best[0] >= .5:
            matched[best[1]]=(pi,prediction,best[0])
    arrays=raw_rows(raw, transform)
    for ti,target in enumerate(truth):
        match=matched.get(ti)
        row={
            'image_index':image_index,'image':record['image'],'image_sha256':record['image_sha256'],
            'label':record['label'],'label_sha256':record['label_sha256'],
            'session_id':record['session_id'],'sequence':record['sequence'],'map_id':record['map_id'],
            'phase':record['phase'],'target_index':ti,'class_id':target.class_id,'target_bbox':list(target.bbox),
            'matched':match is not None,
            'matched_prediction':None if match is None else {'index':match[0],'bbox':list(match[1].bbox),'confidence':match[1].confidence,'iou':match[2]},
        }
        if target.class_id=='platform':
            row['raw_diagnosis']=raw_diagnosis(target,arrays,model.manifest.class_names)
        ledger.append(row)
        if first_miss is None and target.class_id=='platform' and match is None:
            first_miss={'record':record,'image_index':image_index,'missed_target_index':ti,'ground_truth':[asdict(t) for t in truth],'candidate_predictions':[asdict(p) for p in predictions]}

def summary(rows):
    return {'gt':len(rows),'matched':sum(row['matched'] for row in rows),'missed':sum(not row['matched'] for row in rows)}

def grouped(rows, field):
    groups=defaultdict(list)
    for row in rows:
        groups[str(row[field])].append(row)
    return {key:summary(value) for key,value in sorted(groups.items())}

def distribution(values):
    finite=[float(v) for v in values if v is not None]
    return {'samples':len(finite),'min':min(finite) if finite else None,'p50':float(np.percentile(finite,50)) if finite else None,'p95':float(np.percentile(finite,95)) if finite else None,'max':max(finite) if finite else None}
