"""Draw a precise fixture evidence plot; no inference or data/model changes."""
import hashlib
import json
from pathlib import Path

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from matplotlib.font_manager import FontProperties
from matplotlib.lines import Line2D
from matplotlib.patches import Rectangle
from PIL import Image

ROOT = Path(__file__).resolve().parents[2]
BASE = ROOT / 'artifacts/v03-experiment'
first = json.loads((BASE/'diagnosis-first-miss.json').read_text())
source = BASE/'dataset'/first['record']['image']
assert hashlib.sha256(source.read_bytes()).hexdigest()==first['record']['image_sha256']
output = BASE/'diagnosis-first-miss-comparison.png'
metadata = BASE/'diagnosis-first-miss-comparison.json'
if output.exists() or metadata.exists():
    raise FileExistsError('comparison artifacts already exist')
with Image.open(source) as image:
    raster = image.convert('RGB').copy()
regular = FontProperties(fname='/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc')
bold = FontProperties(fname='/usr/share/fonts/opentype/noto/NotoSansCJK-Bold.ttc')
truth=[item for item in first['ground_truth'] if item['class_id']=='platform']

def iou(a,b):
    intersect=max(0,min(a[2],b[2])-max(a[0],b[0]))*max(0,min(a[3],b[3])-max(a[1],b[1]))
    aa=(a[2]-a[0])*(a[3]-a[1]); ab=(b[2]-b[0])*(b[3]-b[1])
    return intersect/(aa+ab-intersect)

def matches(predictions):
    matched=set()
    for item in sorted(predictions,key=lambda p:-p['confidence']):
        candidates=[(iou(item['bbox'],target['bbox']),idx) for idx,target in enumerate(truth) if idx not in matched]
        best=max(candidates,key=lambda pair:(pair[0],-pair[1]),default=None)
        if best is not None and best[0]>=.5: matched.add(best[1])
    return matched

fig,axes=plt.subplots(1,3,figsize=(13.8,4.6),dpi=140)
fig.patch.set_facecolor('#0c1423')
fig.subplots_adjust(left=.025,right=.985,top=.81,bottom=.22,wspace=.06)
colors={'gt':'#39d8ff','pred':'#b5f67c','miss':'#ff71a6'}
titles=['真值：3 个平台','原 v0.2 模型','v0.3 候选：未通过准入']
sets=[truth,[p for p in first['baseline_predictions'] if p['class_id']=='platform'],[p for p in first['candidate_predictions'] if p['class_id']=='platform']]
counts=[]
for column,(ax,title,items) in enumerate(zip(axes,titles,sets)):
    ax.imshow(raster,interpolation='nearest',origin='upper',extent=(0,320,192,0))
    ax.set_xlim(-2,322); ax.set_ylim(194,-2); ax.axis('off')
    ax.set_title(title,fontproperties=bold,fontsize=12,color='#f1f5fc',pad=12)
    for item in items:
        x1,y1,x2,y2=item['bbox']
        color=colors['gt'] if column==0 else colors['pred']
        ax.add_patch(Rectangle((x1,y1),x2-x1,y2-y1,fill=False,edgecolor=color,linewidth=1.4))
        if column:
            ax.text(max(2,x1+2),y1-5,f"{item['confidence']:.2f}",fontsize=8.5,color='#d9ffb9',
                    bbox={'facecolor':'#0b1723','alpha':.88,'edgecolor':'none','pad':1.4})
    found=set(range(len(truth))) if column==0 else matches(items)
    for index,item in enumerate(truth):
        if index in found: continue
        x1,y1,x2,y2=item['bbox']
        ax.add_patch(Rectangle((x1,y1),x2-x1,y2-y1,fill=False,edgecolor=colors['miss'],linewidth=1.7,linestyle=(0,(3,2))))
        ax.text(max(2,x1+2),y1-5,'漏检',fontproperties=bold,fontsize=8.5,color=colors['miss'],
                bbox={'facecolor':'#0b1723','alpha':.9,'edgecolor':'none','pad':1.4})
    counts.append({'panel':title,'platforms_shown':len(items),'matched_ground_truth':len(found)})
fig.text(.025,.925,'同一张验证图：长地板恢复，上层平台遗漏',fontproperties=bold,fontsize=18,color='#f8fbff')
fig.text(.025,.865,'val / meadow / spawn / s0002_00000  ·  仅展示平台检测框',fontproperties=regular,fontsize=10,color='#b5c3d8')
legend=[Line2D([0],[0],color=colors['gt'],lw=2,label='真值'),Line2D([0],[0],color=colors['pred'],lw=2,label='保留的模型预测'),Line2D([0],[0],color=colors['miss'],lw=2,ls='--',label='未匹配真值')]
fig.legend(handles=legend,loc='lower left',bbox_to_anchor=(.019,.09),ncol=3,frameon=False,
           prop=regular,labelcolor='#dce6f7',handlelength=2.5,columnspacing=2.5)
fig.text(.025,.065,'冻结阈值：置信度 0.25，NMS IoU 0.45，匹配 IoU 0.50。该图是首张漏检 val 图，不代表全量成绩。',
         fontproperties=regular,fontsize=9,color='#b5c3d8')
fig.savefig(output,dpi=140,facecolor=fig.get_facecolor(),pil_kwargs={'compress_level':9,'optimize':True})
plt.close(fig)
assert output.stat().st_size<=200_000, output.stat().st_size
record={'schema_version':1,'scope':'validation_fixture_evidence_illustration_not_benchmark','image':output.name,
        'image_sha256':hashlib.sha256(output.read_bytes()).hexdigest(),'bytes':output.stat().st_size,
        'source_image':first['record']['image'],'source_sha256':first['record']['image_sha256'],
        'source_metadata_sha256':hashlib.sha256((BASE/'diagnosis-first-miss.json').read_bytes()).hexdigest(),
        'selection_rule':first['selection_rule'],'panels':counts,'additional_inference_calls':0}
with metadata.open('x',encoding='utf-8',newline='\n') as stream:
    json.dump(record,stream,ensure_ascii=False,indent=2); stream.write('\n')
print(json.dumps(record,ensure_ascii=False,indent=2))
